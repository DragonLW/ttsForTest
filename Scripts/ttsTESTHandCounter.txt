function onObjectEnterZone(enterZoneOper, obj)
    if enterZoneOper == zoneOper then
        if string.find(obj.getName(), "Info") then
            counterOper = counterOper + 1
            counterTotal = counterTotal + 1
        else
            counterTotal = counterTotal + 1
        end
        updateCounter()
    end
end
function onObjectLeaveZone(leaveZoneOper, obj)
    if leaveZoneOper == zoneOper then
        if string.find(obj.getName(), "Info") then
            counterOper = counterOper - 1
            counterTotal = counterTotal - 1
        else
            counterTotal = counterTotal - 1
        end
        updateCounter()
    end
end
