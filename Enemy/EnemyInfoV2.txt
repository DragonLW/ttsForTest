function hp1(obj,holo,alt_click)
    if alt_click == false then
        hp = hp + 1
        self.editButton({index = 0, label = "HP: " ..hp})
    else
        if alt_click == true then
            hp = hp - 1
            self.editButton({index = 0, label = "HP: " ..hp})
        end
    end

    if hp > hps then
        self.editButton({index = 0, font_color = {0,1,0}})
    else
        if hp < hps then
            self.editButton({index = 0, font_color = {1,0,0}})
        else
            if hp == hps then
                self.editButton({index = 0, font_color = {0,0.6,1}})
            end
        end
    end
    if hp <= 0 then
        self.editButton({index = 6, scale = {1,1,1}})
    else
        self.editButton({index = 6, scale = {0,0,0}})
    end
end

function sell(obj, holo, alt_click)
rLmdR = rLmd.getVar("lmdR")
bLmdR = bLmd.getVar("lmdR")
gLmdR = gLmd.getVar("lmdR")
yLmdR = yLmd.getVar("lmdR")
    for i, object in ipairs(Global.getObjects()) do
        if holo == "Red" then
            if object.getName() == "R_LMD" then
                rLmd.setVar("lmdR", rLmdR + sellEn)
                rLmdS = rLmd.getVar("lmdR")
                rLmd.editButton({index = 0, label = rLmdS})
                self.destroyObject()
            end
        else
            if holo == "Blue" then
                if object.getName() == "B_LMD" then
                    bLmd.setVar("lmdR", bLmdR + sellEn)
                    bLmdS = bLmd.getVar("lmdR")
                    bLmd.editButton({index = 0, label = bLmdS})
                    self.destroyObject()
                end
            else
                if holo == "Green" then
                    if object.getName() == "G_LMD" then
                        gLmd.setVar("lmdR", gLmdR + sellEn)
                        gLmdS = gLmd.getVar("lmdR")
                        gLmd.editButton({index = 0, label = gLmdS})
                        self.destroyObject()
                    end
                else
                    if holo == "Yellow" then
                        if object.getName() == "Y_LMD" then
                            yLmd.setVar("lmdR", yLmdR + sellEn)
                            yLmdS = yLmd.getVar("lmdR")
                            yLmd.editButton({index = 0, label = yLmdS})
                            self.destroyObject()
                        end
                    end
                end
            end
        end
    end
end
